# PaintOmics 4 runtime dependencies.
#
# This is the single source of truth. PaintomicsServer/src/requirements.txt was a
# second, conflicting copy (CairoSVG 2.4.2 vs 2.7.0, Pillow 8.0.1 vs 10.3.0) and
# now just points here.
#
# Versions are pinned to the working local environment (Python 3.9.23) so the
# container reproduces a configuration that is known to run, not merely one that
# resolves.
#
# Python must stay at 3.9: Flask 1.1.2 / Werkzeug 1.0.1 / Jinja2 2.11.2 /
# MarkupSafe 1.1.1 do not import on 3.10+.

# --- Web framework (do not bump; see above) ---
Flask==1.1.2
Werkzeug==1.0.1
Jinja2==2.11.2
MarkupSafe==1.1.1
itsdangerous==1.1.0
click==7.1.2

# --- WSGI server ---
# processes = 1 is mandatory: src/common/PySiQ.py holds the job queue in process
# memory, so a second process would get its own empty queue. Concurrency comes
# from threads.
uWSGI==2.0.28

# --- Database ---
# pymongo 4 is required, not optional: the deployment runs MongoDB 7, and
# pymongo 3.11 speaks wire-protocol opcodes MongoDB removed in 5.1. See
# src/tests/test_pymongo4_compat.py.
pymongo==4.10.1

# --- Scientific stack ---
numpy==1.26.4
pandas==1.5.3
scipy==1.13.1
statsmodels==0.14.6
patsy==1.0.2

# --- Image and SVG rendering ---
# pycairo is deliberately absent: nothing imports it, it is not installed in the
# working environment, and it needs cairo headers to build. cairosvg renders via
# cairocffi, which needs the libcairo2 shared library at runtime -- the container
# must install it.
CairoSVG==2.7.0
cairocffi==1.6.0
Pillow==10.3.0
tinycss2==1.0.2
cssselect2==0.3.0
webencodings==0.5.1
defusedxml==0.6.0
cffi==1.15.1
pycparser==2.21

# --- HTTP ---
# requests is imported by AdminTools/DBManager.py and scripts/downloadReactome.py
# but was pinned in neither requirements file.
requests==2.32.5

# --- AI interpretation pipeline (src/classes/AIInterpret/) ---
openai==2.25.0
openai-agents==0.8.4
# Required on Python 3.9, and not declared by openai-agents. Its models use PEP
# 604 annotations ("float | None"), which pydantic cannot evaluate before 3.10
# without this backport -- `import agents` raises TypeError at import time.
# The local conda environment only worked because something else pulled it in
# transitively; a clean install of this file did not.
eval_type_backport==0.3.1

# --- Scheduling and utilities ---
